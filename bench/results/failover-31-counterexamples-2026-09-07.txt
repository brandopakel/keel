func TestReviewSuccessorPromotionMustFenceIsolatedPrimary(t *testing.T) {
	setupFailover(t)
	require.Equal(t, "OK", run(t, "KEEL.PROMOTE", "1"))
	isolatedPrimary := failover
	// A separate node is promoted without any delivery to the isolated one.
	require.NoError(t, LoadTerm(filepath.Join(t.TempDir(), "successor.aof")))
	require.Equal(t, "OK", run(t, "KEEL.PROMOTE", "2"))
	require.True(t, Writable())
	failover = isolatedPrimary
	require.False(t, Writable(), "the isolated old primary still accepts writes alongside its successor")
}

func TestReviewFenceMustSurvivePrimaryRestart(t *testing.T) {
	path := setupFailover(t)
	require.Equal(t, "OK", run(t, "KEEL.PROMOTE", "1"))
	require.Equal(t, "OK", run(t, "KEEL.FENCE", "2"))
	require.False(t, Writable())
	require.NoError(t, CloseAOF())
	require.NoError(t, LoadTerm(path))
	require.False(t, Writable(), "restart must not turn the observed successor term into this node's held authority")
}
